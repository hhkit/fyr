  \begin{algorithmic}
  \ForAll{$i,j \in [0,N) \times [0,N)$}
    \For {$k \in [0,d)$}
      \State $S_{i,j} := S_{i,j} + Q_{i,k} K_{j,k}$
    \EndFor
  \EndFor
  \ForAll{$i \in [0,N)$}
    \For {$j \in [0,N)$}
      \State $\tilde{m} = \max(m_{i}, S_{i,j})$
      \State $\tilde{P} = e^{S_{i,j} - \tilde{m}}$
      \State $\tilde{\alpha} = e^{m_i - \tilde{m}} $
      \State $m_i := \tilde{m}$
      \State $\ell_i := \alpha \ell_i + \tilde{P}$
      \ForAll {$k \in [0,d)$}
        \State $O_{i,k} := \alpha O_{i,k} + \tilde{P} V_{j,k}$
      \EndFor
    \EndFor
  \EndFor
  \ForAll{$i,j \in [0,N) \times [0,d)$}
    \State $O_{i,j} := O_{i,j} / \ell_i$
  \EndFor
\end{algorithmic}
